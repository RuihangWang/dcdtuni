891: inline bool qualityLineage
892: (
893:     const Stencil& stencil,const State& before,const State& after,scalar ceiling
894: )
895: {
896:     if (!before.valid || !after.valid || !finiteScalar(ceiling)) return false;
897:     auto compare=[](const std::vector<scalar>& a,const std::vector<scalar>& b,scalar limit,bool upper)
898:     {
899:         return dcCartFiniteRepairKernel::preservesHardRoles(a,b,limit,upper,scalar(VSMALL));
900:     };
901:     // Include outside determinants as roles; unchanged geometry is additionally
902:     // checked bitwise below rather than inferred from the quality verdict.
903:     if (!compare(before.determinant,after.determinant,.001,false)) return false;
904:     for (label k=0;k<3;++k)
905:     {
906:         std::vector<scalar> a,b;
907:         for (label f=0;f<label(stencil.faces.size());++f) if (!stencil.faces[f].coupled)
908:         { a.push_back(before.faceQuality[f][k]);b.push_back(after.faceQuality[f][k]); }
909:         if (!compare(a,b,k==0 ? .05 : k==1 ? .01 : 4,k==2)) return false;
910:     }
911:     label oldN=0,newN=0;
912:     for (label f=0;f<label(stencil.faces.size());++f) if (!stencil.faces[f].coupled)
913:     {
914:         oldN+=before.faceQuality[f][3]>70;newN+=after.faceQuality[f][3]>70;
915:         if (after.faceQuality[f][3]>ceiling) return false;
916:     }
917:     if (newN>oldN) return false;
918:     for (label c:stencil.outside)
919:     {
920:         if (!sameVector(before.cc[c],after.cc[c])
921:          || !sameScalar(before.volume[c],after.volume[c])) return false;
922:         for (label f:stencil.cells[c].faces)
923:             if (!sameVector(before.sf[f],after.sf[f])
924:              || !sameVector(before.fc[f],after.fc[f])) return false;
925:     }
926:     return true;
927: }

937:     const scalar planeTol=1e-10*stencil.scale;
938:     for (label ci:stencil.core)
939:     {
940:         const CellData& c=stencil.cells[ci];
941:         oldVolume+=before.volume[ci];newVolume+=after.volume[ci];
942:         oldMoment+=before.volume[ci]*before.cc[ci];
943:         newMoment+=after.volume[ci]*after.cc[ci];
944:         vector closure(Zero);scalar area=0;
945:         for (label f:c.faces)
946:         {
947:             const scalar sense=stencil.faces[f].owner==c.id ? 1 : -1;
948:             const vector normal=sense*after.sf[f]/mag(after.sf[f]);
949:             closure+=sense*after.sf[f];area+=mag(after.sf[f]);

951:             for (label g:c.faces) for (label vertex:stencil.faces[g].vertices)
952:                 if ((normal & (points[vertex]-after.fc[f]))>planeTol) return false;
953:             for (label g:c.faces) if (g!=f)
954:             {
955:                 const vector delta=after.fc[g]-after.fc[f];
956:                 if ((normal & (delta/max(mag(delta),VSMALL)))>-1e-6) return false;
957:             }

959:             const face& poly=stencil.faces[f].vertices;
960:             for (label v:poly)
961:                 if (mag(normal & (points[v]-after.fc[f]))>planeTol) return false;

967:                         < -planeTol*mag(after.sf[f])*max(mag(edge),ROOTVSMALL))
968:                         return false;

982:     for (label f=0;f<label(stencil.faces.size());++f)
983:     {

988:         if (owner!=stencil.cellIndex.end())
989:             touchesCore=std::find(stencil.core.begin(),stencil.core.end(),owner->second)!=stencil.core.end();
990:         if (neighbour!=stencil.cellIndex.end())
991:             touchesCore=touchesCore || std::find(stencil.core.begin(),stencil.core.end(),neighbour->second)!=stencil.core.end();
992:         if (!touchesCore || fd.coupled) continue;
993:         const face& poly=fd.vertices;bool shared=false;
994:         forAll(poly,anchor)
995:         {
996:             bool good=true;
997:             const auto own=stencil.cellGeometry(fd.owner,after);
998:             for (label j=1;j<poly.size()-1;++j)
999:             {
1000:                 label a=poly[(anchor+j)%poly.size()],b=poly[(anchor+j+1)%poly.size()];
1001:                 scalar q=tetPointRef(own.first,points[poly[anchor]],points[a],points[b]).quality();
1002:                 if (!finiteScalar(q) || q<=polyMeshTetDecomposition::minTetQuality) { good=false;break; }
1003:                 if (fd.neighbour>=0)
1004:                 {
1005:                     const auto nei=stencil.cellGeometry(fd.neighbour,after);
1006:                     q=tetPointRef(nei.first,points[poly[anchor]],points[b],points[a]).quality();
1007:                     if (!finiteScalar(q) || q<=polyMeshTetDecomposition::minTetQuality) { good=false;break; }
1008:                 }
1009:             }
1010:             if (good) { shared=true;break; }
1011:         }
