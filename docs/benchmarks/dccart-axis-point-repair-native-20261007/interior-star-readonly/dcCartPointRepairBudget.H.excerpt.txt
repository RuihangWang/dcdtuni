1: // Optional native repair reserves time for the remaining product pipeline.
2: #ifndef dcCartPointRepairBudget_H
3: #define dcCartPointRepairBudget_H
4: 
5: #include <algorithm>
6: #include <cmath>
7: #include <cstring>
8: 
9: namespace Foam
10: {
11: namespace dcCartPointRepairBudget
12: {
13: inline bool enabled(const char* flag)
14: {
15:     return flag && std::strcmp(flag, "1") == 0;
16: }
17: 
18: inline double allowance(double runLimitSeconds, double runElapsedSeconds)
19: {
20:     constexpr double stageCap = 20;
21:     constexpr double downstreamReserve = 60;
22:     if (!std::isfinite(runLimitSeconds) || runLimitSeconds < 0
23:      || !std::isfinite(runElapsedSeconds) || runElapsedSeconds < 0)
24:     {
25:         return 0;
26:     }
27:     if (runLimitSeconds == 0) { return stageCap; }
28:     return std::min(stageCap,
29:         std::max(0.0, runLimitSeconds-runElapsedSeconds-downstreamReserve));
30: }
31: }
32: }
33: #endif
