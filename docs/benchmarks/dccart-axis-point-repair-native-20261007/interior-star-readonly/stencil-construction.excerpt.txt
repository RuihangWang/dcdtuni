586:                     {
587:                         if (fixed<0) fixed=p;
588:                         else level=level && same(points[fixed][a],points[p][a]);
589:                     }
590:                     if (fixed>=0 && level
591:                      && !same(points[fixed][a],points[pair[0]][a])) allowed=true;
592:                 }
593:             }
594:             if (!allowed || !valid) { valid=false; break; }
595:         }
596:         if (valid && polygon)
597:         {
598:             out.axis=axis; out.pairA=pair[0]; out.pairB=pair[1];
599:             out.dimension=4; return out;
600:         }
601:     }
602:     return out;
603: }
