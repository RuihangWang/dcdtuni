615:         # If neither options are enabled, then do nothing to the Viewport's resolution
616:         if not self.__push_ui_to_viewport and not self.__expand_viewport_to_ui:
617:             return False

619:         # Match the legacy Viewport resolution and fit
620:         # XXX: Note for a down-sampled constant resolution, that is expanded to the Viewport dimensions
621:         # the order of operation is off if one expects 512 x 512 @ 1 to 1024 x 1024 @ 0.5 to expand to the same result
622: 
623:         # Get the omni.ui.Widget computed size
624:         ui_frame = self.__ui_frame
625:         res_x, res_y = int(ui_frame.computed_width), int(ui_frame.computed_height)
626:         # Scale by DPI as legacy Viewport (to match legacy Viewport calculations)
627:         dpi_scale = ui.Workspace.get_dpi_scale() if self.__resolution_uses_dpi else 1.0
628:         res_x, res_y = res_x * dpi_scale, res_y * dpi_scale
629:         # Full resolution width and hight
630:         fres_x, fres_y = self.__full_resolution if self.__full_resolution else (res_x, res_y)
631:         if self.__expand_viewport_to_ui and (fres_y > 0) and (res_y > 0):
632:             tex_ratio = fres_x / fres_y
633:             ui_ratio = res_x / res_y
634:             if tex_ratio < ui_ratio:
635:                 fres_x = fres_x * (ui_ratio / tex_ratio)
636:             else:
637:                 fres_y = fres_y * (tex_ratio / ui_ratio)
638: 
639:             # Limit the resolution to not grow too large
