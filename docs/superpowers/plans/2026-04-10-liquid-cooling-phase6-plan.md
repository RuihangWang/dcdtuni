**Goal:** Enable manual editing of liquid cooling pipe routing at two levels: (1) server-internal cold plate drag-connect wiring in Equipment Level View, (2) CDU-Rack 3D waypoint pipe editing with component insertion.

**Tech Stack:** Canvas 2D (Part A), Three.js r128 TransformControls + TubeGeometry (Part B), Python Pydantic v2 + NetworkX (backend)

```python
def _build_from_chains(G, components, liquid_cfg, fluid, specs):
    """Build network from explicit piping chains (manual mode)."""
    # Add CDU boundary nodes
    for cdu_name, cdu_cfg in liquid_cfg.cdus.items():
        supply_node = f"{cdu_name}_supply"
        return_node = f"{cdu_name}_return"
        estimated_supply_kPa = max(50.0, cdu_cfg.totalFlowRate * 3.0)
        G.add_node(supply_node, pressure_fixed=estimated_supply_kPa)
        G.add_node(return_node, pressure_fixed=0.0)

    for chain in liquid_cfg.piping.chains:
        path = chain.path
        if not path:
            continue

        # Parse alternating: port_str, component_dict, port_str, component_dict, ...
        i = 0
        while i < len(path):
            item = path[i]

            # Port string — ensure node exists
            if isinstance(item, str):
                if not G.has_node(item):
                    G.add_node(item)
                i += 1
                continue

            # Component dict — must be between two port strings
            if isinstance(item, dict):
                # Find prev and next port strings
                prev_port = None
                for j in range(i - 1, -1, -1):
                    if isinstance(path[j], str):
                        prev_port = path[j]
                        break
                next_port = None
                for j in range(i + 1, len(path)):
                    if isinstance(path[j], str):
                        next_port = path[j]
                        break

                if prev_port is None or next_port is None:
                    i += 1
                    continue

                # Ensure nodes exist
                if not G.has_node(prev_port):
                    G.add_node(prev_port)
                if not G.has_node(next_port):
                    G.add_node(next_port)

                # Create component based on type
                comp_name = f"{chain.id}:step{i}"
                component = _create_chain_component(item, specs, fluid)
                if component is not None:
                    G.add_edge(prev_port, next_port)
                    components.append(ComponentRecord(comp_name, prev_port, next_port, component))

                i += 1
                continue

            i += 1


def _create_chain_component(item, specs, fluid):
    """Create a component nn.Module from a chain segment dict."""
    from dcdtuni.liquid.factory import create_component_by_name

    if "pipe" in item:
        wp = item.get("waypoints", [])
        # Estimate pipe length from waypoints or use default
        length = item.get("length")
        if length is None and len(wp) >= 2:
            total = 0.0
            for k in range(1, len(wp)):
                p0 = wp[k - 1] if isinstance(wp[k - 1], dict) else {"x": 0, "y": 0, "z": 0}
                p1 = wp[k] if isinstance(wp[k], dict) else {"x": 0, "y": 0, "z": 0}
                dx = p1.get("x", 0) - p0.get("x", 0)
                dy = p1.get("y", 0) - p0.get("y", 0)
                dz = p1.get("z", 0) - p0.get("z", 0)
                total += (dx**2 + dy**2 + dz**2) ** 0.5
            length = max(total, 0.5)
        return create_component_by_name("pipe", item["pipe"], specs, fluid.mu, fluid.rho,
                                        pipe_length=length)
    elif "valve" in item:
        comp = create_component_by_name("valve", item["valve"], specs, fluid.mu, fluid.rho)
        if hasattr(comp, "set_opening") and "opening" in item:
            comp.set_opening(item["opening"])
        return comp
    elif "pump" in item:
        comp = create_component_by_name("pump", item["pump"], specs, fluid.mu, fluid.rho)
        if hasattr(comp, "adjust_speed") and "rpm" in item:
            comp.adjust_speed(item["rpm"])
        return comp
    elif "cold_plate" in item:
        return create_component_by_name("cold_plate", item["cold_plate"], specs, fluid.mu, fluid.rho)
    elif "quick_connect" in item:
        return create_component_by_name("quick_connect", item["quick_connect"], specs, fluid.mu, fluid.rho)
    elif "heat_exchanger" in item:
        return create_component_by_name("heat_exchanger", item["heat_exchanger"], specs, fluid.mu, fluid.rho)
    return None
```

  function onCanvasMouseDown(e) {
    var pos = getCanvasPos(e);

  function onCanvasMouseUp(e) {
    if (!_dragFrom) return;
    var pos = getCanvasPos(e);
    var hitPort = hitTestPort(pos.x, pos.y, 'in');
    if (hitPort && hitPort.eqId === _dragFrom.eqId) {
      // Valid connection: outlet → inlet on same equipment
      createConnection(_dragFrom.eqId, _dragFrom.portId, hitPort.portId);
    }
    _dragFrom = null;
    _canvas.style.cursor = 'default';

  function getCanvasPos(e) {
    var rect = _canvas.getBoundingClientRect();
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  }

  function hitTestPort(x, y, typeFilter) {
    var hitRadius = PORT_RADIUS + 8;
    for (var i = 0; i < _portPositions.length; i++) {
      var p = _portPositions[i];
      if (typeFilter && p.type !== typeFilter) continue;
      var dx = x - p.x, dy = y - p.y;
      if (dx * dx + dy * dy < hitRadius * hitRadius) return p;
    }
    return null;
  }

  function hitTestConnection(x, y) {
    // Simple proximity test to bezier connections
    // (approximated as line segment for hit testing)
    var rack = _selectedRackId ? window._modelMap && window._modelMap.get(_selectedRackId) : null;
    if (!rack) return null;
    var equipments = (rack.rack && rack.rack.equipments) || [];
    for (var i = 0; i < equipments.length; i++) {
      var eq = equipments[i];
      if (!eq.liquidCooling || !eq.liquidCooling.routing || eq.liquidCooling.routing.type !== 'custom') continue;
      var conns = eq.liquidCooling.routing.connections || [];
      for (var j = 0; j < conns.length; j++) {
        var c = conns[j];
        var fp = _portPositions.find(function(p) { return p.eqId === eq.id && p.portId === c.from; });
        var tp = _portPositions.find(function(p) { return p.eqId === eq.id && p.portId === c.to; });
        if (fp && tp && distToSegment(x, y, fp.x, fp.y, tp.x, tp.y) < 8) {
          return { eqId: eq.id, fromPort: c.from, toPort: c.to };
        }
      }
    }
    return null;
  }

  function distToSegment(px, py, x1, y1, x2, y2) {
    var dx = x2 - x1, dy = y2 - y1;
    var t = Math.max(0, Math.min(1, ((px - x1) * dx + (py - y1) * dy) / (dx * dx + dy * dy + 1e-12)));
    var nx = x1 + t * dx, ny = y1 + t * dy;
    return Math.sqrt((px - nx) * (px - nx) + (py - ny) * (py - ny));
  }

  function createConnection(eqId, fromPortId, toPortId) {
    var rack = _selectedRackId ? window._modelMap && window._modelMap.get(_selectedRackId) : null;
    if (!rack) return;
    var eq = (rack.rack.equipments || []).find(function(e) { return e.id === eqId; });
    if (!eq || !eq.liquidCooling) return;

    // Validate: no duplicate inlet
    var conns = eq.liquidCooling.routing.connections || [];
    if (conns.some(function(c) { return c.to === toPortId; })) return; // inlet already connected
    if (conns.some(function(c) { return c.from === fromPortId; })) return; // outlet already connected

    if (typeof pushUndoCheckpoint === 'function') pushUndoCheckpoint('createConnection');
    eq.liquidCooling.routing.type = 'custom';
    if (!eq.liquidCooling.routing.connections) eq.liquidCooling.routing.connections = [];
    eq.liquidCooling.routing.connections.push({ from: fromPortId, to: toPortId });
    render();
    if (typeof logOperation === 'function') {
      logOperation('createConnection', 'Connect ' + fromPortId + ' → ' + toPortId, {eqId, fromPortId, toPortId}, 'modify');
    }
  }

  window.deleteSelectedConnection = function() {

  function insertComponentInChain(chainIdx, segmentIdx, compType, specName) {
    if (typeof pushUndoCheckpoint === 'function') pushUndoCheckpoint('insertComponent');
    var chains = state.liquidCooling.piping.chains;
    if (!chains || !chains[chainIdx]) return;
    var path = chains[chainIdx].path;

    // Find the pipe segment at segmentIdx and split it
    var pipeItem = null;
    var pipeIdx = -1;
    var count = 0;
    for (var i = 0; i < path.length; i++) {
      if (typeof path[i] === 'object' && path[i].pipe) {
        if (count === segmentIdx) {
          pipeItem = path[i];
          pipeIdx = i;
          break;
        }
        count++;
      }
    }
    if (!pipeItem || pipeIdx < 0) return;

    // Create intermediate port name
    var junctionName = chains[chainIdx].id + '_junction_' + Date.now();

    // Build component object
    var compObj = {};
    compObj[compType] = specName;
    if (compType === 'valve') compObj.opening = 1.0;
    if (compType === 'pump') compObj.rpm = 3000;

    // Split: original pipe → junction → component → junction2 → pipe
    var junctionName2 = junctionName + '_post';
    var wp = pipeItem.waypoints || [];
    var midIdx = Math.floor(wp.length / 2);
    var wp1 = wp.slice(0, midIdx + 1);
    var wp2 = wp.slice(midIdx);

    // Replace the pipe segment with: pipe1, junction, component, junction2, pipe2
    var newItems = [
      {pipe: pipeItem.pipe, waypoints: wp1},
      junctionName,
      compObj,
      junctionName2,
      {pipe: pipeItem.pipe, waypoints: wp2},
    ];
    path.splice(pipeIdx, 1, ...newItems);

    rebuildEditVisuals();
    updatePipeGeometry();
    if (window.requestRender) window.requestRender();
    if (typeof logOperation === 'function') {
      logOperation('insertComponent', 'Insert ' + compType + ' in chain', {chainIdx, compType, specName}, 'modify');
    }
  }
